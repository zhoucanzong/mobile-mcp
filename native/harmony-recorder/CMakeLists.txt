cmake_minimum_required(VERSION 3.16)
project(mobile_mcp_recorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

option(MOBILE_MCP_HOST_TESTS "Build portable lifecycle tests without the device SDK" OFF)
if(MOBILE_MCP_HOST_TESTS)
    enable_testing()
    add_executable(recorder-session-test session_test.cpp)
    target_link_libraries(recorder-session-test PRIVATE Threads::Threads)
    add_test(NAME recorder-session COMMAND recorder-session-test)
else()
    add_executable(mobile-mcp-recorder main.cpp)
    target_link_libraries(mobile-mcp-recorder PRIVATE native_avscreen_capture Threads::Threads)
endif()
